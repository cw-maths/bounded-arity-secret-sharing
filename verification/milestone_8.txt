  {G : Type u_2} →
    [inst : Fintype P] →
      [DecidableEq P] → (K : Type u_3) → [Field K] → (G → Finset P) → BoundedArity.AccessStructure P → ℕ → Prop :=

  ∃ n s W,
    Module.finrank K ↥(K ∙ s) = 1 ∧
      ∀ (I : Finset G),

            BoundedArity.physicalTable b (BoundedArity.physicalUnion B I) ∧
          Module.finrank K ↥(K ∙ s ⊔ BoundedArity.jointSubspace W I) =
            BoundedArity.physicalTable b (BoundedArity.physicalUnion B I) +
              BoundedArity.unauthorizedBit Γ (BoundedArity.physicalUnion B I)

  [inst : Fintype P] → [DecidableEq P] → ℕ → BoundedArity.AccessStructure P → (Finset (Option P) → ℝ) → Prop :=

  BoundedArity.IsPolymatroid F ∧
    BoundedArity.HasNormalizedAccess Γ F ∧
      ∀ k ≤ r,
        ∀ (a : Finset (Fin k) → ℝ),
          BoundedArity.ValidInformation a →
            ∀ (A : Fin k → Finset (Option P)), 0 ≤ BoundedArity.inequalityValue a (BoundedArity.groupProjection F A)
